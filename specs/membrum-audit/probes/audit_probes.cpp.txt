// ---------------------------------------------------------------- printing
std::string fmt(double v, int prec = 2)
{
    char b[64];
    std::snprintf(b, sizeof(b), "%.*f", prec, v);
    return b;
}

void probe(const std::string& name, const std::string& setup, const std::string& metric,
           const std::string& value, const std::string& thr, const char* verdict)
{
    std::printf("PROBE %s | %s | %s = %s | threshold %s | %s\n", name.c_str(), setup.c_str(),
                metric.c_str(), value.c_str(), thr.c_str(), verdict);
    std::fflush(stdout);
}

// ---------------------------------------------------------------- analysis
double toDb(double lin) { return 20.0 * std::log10(std::max(lin, 1e-12)); }

struct Stats
{
    double peak = 0.0;
    double rms = 0.0;
    double dc = 0.0;
    int nonFinite = 0;
};

Stats stats(const float* x, std::size_t n)
{
    Stats s;
    double sq = 0.0;
    double sum = 0.0;
    std::size_t cnt = 0;
    for (std::size_t i = 0; i < n; ++i)
    {
        const float v = x[i];
        if (!Krate::DSP::detail::isFinite(v)) { ++s.nonFinite; continue; }
        s.peak = std::max(s.peak, static_cast<double>(std::fabs(v)));
        sq += static_cast<double>(v) * v;
        sum += v;
        ++cnt;
    }
    if (cnt > 0)
    {
        s.rms = std::sqrt(sq / static_cast<double>(cnt));
        s.dc = sum / static_cast<double>(cnt);
    }
    return s;
}
Stats stats(const Buf& b, std::size_t from = 0, std::size_t to = SIZE_MAX)
{
    to = std::min(to, b.size());
    if (from >= to) return {};
    return stats(b.data() + from, to - from);
}

// Time (s) until the 10 ms RMS envelope stays 60 dB below its maximum. -1 = never.
double decay60(const Buf& x, double sr, double belowDb = 60.0)
{
    const std::size_t win = static_cast<std::size_t>(sr * 0.01);
    const std::size_t nw = x.size() / win;
    if (nw == 0) return -1.0;
    std::vector<double> r(nw);
    double mx = 0.0;
    for (std::size_t w = 0; w < nw; ++w)
    {
        double sq = 0.0;
        for (std::size_t i = 0; i < win; ++i)
        {
            const float v = x[w * win + i];
            if (Krate::DSP::detail::isFinite(v)) sq += static_cast<double>(v) * v;
        }
        r[w] = std::sqrt(sq / static_cast<double>(win));
        mx = std::max(mx, r[w]);
    }
    if (mx <= 0.0) return 0.0;
    const double thr = mx * std::pow(10.0, -belowDb / 20.0);
    std::size_t last = 0;
    for (std::size_t w = 0; w < nw; ++w)
        if (r[w] > thr) last = w;
    if (last + 1 >= nw) return -1.0;
    return static_cast<double>(last + 1) * static_cast<double>(win) / sr;
}

std::string decayStr(double d, double maxS)
{
    return d < 0.0 ? (">" + fmt(maxS, 1) + " s") : (fmt(d, 3) + " s");
}

// Last time (s) the absolute sample exceeds `thrLin`. -1 = still above at the end.
double lastAbove(const Buf& x, double sr, double thrLin)
{
    std::size_t last = 0;
    bool any = false;
    for (std::size_t i = 0; i < x.size(); ++i)
        if (std::fabs(x[i]) > thrLin) { last = i; any = true; }
    if (!any) return 0.0;
    if (last + static_cast<std::size_t>(sr * 0.05) >= x.size()) return -1.0;
    return static_cast<double>(last) / sr;
}

std::vector<double> hann(std::size_t n)
{
    std::vector<double> w(n);
    for (std::size_t i = 0; i < n; ++i)
        w[i] = 0.5 - 0.5 * std::cos(2.0 * std::numbers::pi * static_cast<double>(i) / static_cast<double>(n - 1));
    return w;
}

// Magnitude spectrum (Hann) of n samples starting at x; returns bins 0..N/2.
std::vector<double> magSpectrum(const float* x, std::size_t n, std::size_t& N)
{
    N = 8192;
    while (N > n && N > 256) N /= 2;
    std::vector<double> mag;
    if (n < 256) return mag;
    Krate::DSP::FFT fft;
    fft.prepare(N);
    std::vector<float> in(N);
    std::vector<Krate::DSP::Complex> out(N / 2 + 1);
    const auto w = hann(N);
    for (std::size_t i = 0; i < N; ++i)
    {
        const float v = x[i];
        in[i] = Krate::DSP::detail::isFinite(v) ? static_cast<float>(v * w[i]) : 0.0f;
    }
    fft.forward(in.data(), out.data());
    mag.resize(N / 2 + 1);
    for (std::size_t k = 0; k <= N / 2; ++k)
        mag[k] = std::sqrt(static_cast<double>(out[k].real) * out[k].real + static_cast<double>(out[k].imag) * out[k].imag);
    return mag;
}

double centroid(const Buf& x, double sr, std::size_t start, std::size_t len)
{
    if (start >= x.size()) return 0.0;
    len = std::min(len, x.size() - start);
    std::size_t N = 0;
    const auto m = magSpectrum(x.data() + start, len, N);
    if (m.empty()) return 0.0;
    double num = 0.0;
    double den = 0.0;
    for (std::size_t k = 1; k < m.size(); ++k)
    {
        const double f = static_cast<double>(k) * sr / static_cast<double>(N);
        num += f * m[k];
        den += m[k];
    }
    return den > 0.0 ? num / den : 0.0;
}

// Fraction (dB) of spectral energy above fLo in a window.
double energyAboveDb(const Buf& x, double sr, std::size_t start, std::size_t len, double fLo)
{
    if (start >= x.size()) return -200.0;
    len = std::min(len, x.size() - start);
    std::size_t N = 0;
    const auto m = magSpectrum(x.data() + start, len, N);
    if (m.empty()) return -200.0;
    double hi = 0.0;
    double tot = 0.0;
    for (std::size_t k = 1; k < m.size(); ++k)
    {
        const double f = static_cast<double>(k) * sr / static_cast<double>(N);
        tot += m[k] * m[k];
        if (f >= fLo) hi += m[k] * m[k];
    }
    return tot > 0.0 ? 10.0 * std::log10(std::max(hi / tot, 1e-20)) : -200.0;
}

// Hann-windowed DTFT magnitude at f.
double dtftMag(const float* x, const std::vector<double>& w, double f, double sr)
{
    const std::size_t n = w.size();
    const double dph = 2.0 * std::numbers::pi * f / sr;
    const double c = std::cos(dph);
    const double s = std::sin(dph);
    double re = 1.0;
    double im = 0.0;
    double ar = 0.0;
    double ai = 0.0;
    for (std::size_t i = 0; i < n; ++i)
    {
        const double v = Krate::DSP::detail::isFinite(x[i]) ? static_cast<double>(x[i]) * w[i] : 0.0;
        ar += v * re;
        ai -= v * im;
        const double nr = re * c - im * s;
        im = re * s + im * c;
        re = nr;
    }
    return std::sqrt(ar * ar + ai * ai);
}

// Best peak within +-cents of fGuess (grid then golden refinement).
double refinePeak(const float* x, const std::vector<double>& w, double sr, double fGuess,
                  double cents, double* magOut = nullptr)
{
    const int kGrid = 61;
    double bestF = fGuess;
    double bestM = -1.0;
    for (int g = 0; g < kGrid; ++g)
    {
        const double c = -cents + 2.0 * cents * static_cast<double>(g) / (kGrid - 1);
        const double f = fGuess * std::pow(2.0, c / 1200.0);
        if (f <= 0.0 || f >= sr * 0.5) continue;
        const double m = dtftMag(x, w, f, sr);
        if (m > bestM) { bestM = m; bestF = f; }
    }
    const double step = 2.0 * cents / (kGrid - 1);
    double lo = bestF * std::pow(2.0, -step / 1200.0);
    double hi = bestF * std::pow(2.0, step / 1200.0);
    const double gr = 0.6180339887;
    for (int it = 0; it < 30; ++it)
    {
        const double a = hi - gr * (hi - lo);
        const double b = lo + gr * (hi - lo);
        if (dtftMag(x, w, a, sr) > dtftMag(x, w, b, sr)) hi = b; else lo = a;
    }
    const double f = 0.5 * (lo + hi);
    if (magOut) *magOut = dtftMag(x, w, f, sr);
    return f;
}

// Dominant spectral peak between fLo..fHi, refined.
double dominantPeak(const Buf& x, double sr, std::size_t start, std::size_t len, double fLo, double fHi)
{
    if (start >= x.size()) return 0.0;
    len = std::min(len, x.size() - start);
    std::size_t N = 0;
    const auto m = magSpectrum(x.data() + start, len, N);
    if (m.empty()) return 0.0;
    std::size_t best = 0;
    double bm = -1.0;
    for (std::size_t k = 1; k < m.size(); ++k)
    {
        const double f = static_cast<double>(k) * sr / static_cast<double>(N);
        if (f < fLo || f > fHi) continue;
        if (m[k] > bm) { bm = m[k]; best = k; }
    }
    if (best == 0) return 0.0;
    const double fCoarse = static_cast<double>(best) * sr / static_cast<double>(N);
    const std::size_t n2 = std::min<std::size_t>(len, static_cast<std::size_t>(sr * 0.5));
    const auto w = hann(n2);
    const double cents = 1200.0 * std::log2((fCoarse + 1.5 * sr / static_cast<double>(N)) / fCoarse);
    return refinePeak(x.data() + start, w, sr, fCoarse, std::min(cents, 600.0));
}

double cents(double f, double ref) { return (f > 0.0 && ref > 0.0) ? 1200.0 * std::log2(f / ref) : 0.0; }

double maxJump(const Buf& x, std::size_t from = 0, std::size_t to = SIZE_MAX)
{
    to = std::min(to, x.size());
    double j = 0.0;
    for (std::size_t i = std::max<std::size_t>(from, 1); i < to; ++i)
        j = std::max(j, static_cast<double>(std::fabs(x[i] - x[i - 1])));
    return j;
}

int clickCount(const Buf& x, double sr, std::size_t from = 0, std::size_t to = SIZE_MAX)
{
    to = std::min(to, x.size());
    if (to <= from + 1024) return 0;
    Krate::DSP::TestUtils::ClickDetectorConfig cfg;
    cfg.sampleRate = static_cast<float>(std::min(sr, 192000.0));
    cfg.frameSize = 512;
    cfg.hopSize = 256;
    cfg.detectionThreshold = 5.0f;
    Krate::DSP::TestUtils::ClickDetector det(cfg);
    det.prepare();
    return static_cast<int>(det.detect(x.data() + from, to - from).size());
}

double maxAbsDiff(const Buf& a, const Buf& b)
{
    const std::size_t n = std::min(a.size(), b.size());
    double d = 0.0;
    for (std::size_t i = 0; i < n; ++i)
        d = std::max(d, static_cast<double>(std::fabs(a[i] - b[i])));
    return d;
}

long firstAbove(const Buf& x, double thr, std::size_t from = 0)
{
    for (std::size_t i = from; i < x.size(); ++i)
        if (std::fabs(x[i]) > thr) return static_cast<long>(i);
    return -1;
}

// ---------------------------------------------------------------- voice rig
Buf renderVoice(const PadConfig& cfg, float vel, double sr, long total, int bs = 256,
                long noteOffAt = -1,
                const std::function<void(Membrum::DrumVoice&, long)>& perBlock = {},
                bool* activeAtEnd = nullptr)
{
    auto v = std::make_unique<Membrum::DrumVoice>();
    v->prepare(sr, 0u);
    Membrum::applyPadConfigToVoice(*v, cfg);
    v->noteOn(vel);
    Buf out(static_cast<std::size_t>(total), 0.0f);
    long pos = 0;
    while (pos < total)
    {
        const int n = static_cast<int>(std::min<long>(bs, total - pos));
        if (noteOffAt >= 0 && pos <= noteOffAt && noteOffAt < pos + n) v->noteOff();
        if (perBlock) perBlock(*v, pos);
        v->processBlock(out.data() + pos, n);
        pos += n;
    }
    if (activeAtEnd) *activeAtEnd = v->isActive();
    return out;
}

// ---------------------------------------------------------------- processor rig
struct Ev
{
    long pos;
    int pitch;
    float vel;  // <= 0 -> note-off
};
struct Pc
{
    long pos;
    ParamID id;
    double v;
};

struct Render
{
    std::vector<Buf> L;
    std::vector<Buf> R;
    std::vector<std::uint64_t> mainFlags;  // per block
    std::vector<std::vector<std::uint64_t>> busFlags;
    std::vector<int> activeVoices;          // per block
};

class Host
{
public:
    Host(double sr, int maxBlock, int numBuses = 1) : sr_(sr), maxBlock_(maxBlock), numBuses_(numBuses)
    {
        p = std::make_unique<Membrum::Processor>();
        p->initialize(nullptr);
        ProcessSetup s{};
        s.processMode = kRealtime;
        s.symbolicSampleSize = kSample32;
        s.maxSamplesPerBlock = maxBlock;
        s.sampleRate = sr;
        p->setupProcessing(s);
        p->setActive(true);
    }
    ~Host()
    {
        p->setActive(false);
        p->terminate();
    }
    Host(const Host&) = delete;
    Host& operator=(const Host&) = delete;

    void activate(int bus, bool on) { p->activateBus(kAudio, kOutput, bus, on ? 1 : 0); }

    Render render(long total, int bs, const std::vector<Ev>& evs, const std::vector<Pc>& pcs = {},
                  int nullBus = -1, const std::function<void(long)>& preBlock = {})
    {
        bs = std::min(bs, maxBlock_);
        Render r;
        r.L.assign(static_cast<std::size_t>(numBuses_), Buf(static_cast<std::size_t>(total), 0.0f));
        r.R.assign(static_cast<std::size_t>(numBuses_), Buf(static_cast<std::size_t>(total), 0.0f));
        r.busFlags.resize(static_cast<std::size_t>(numBuses_));
        std::vector<Buf> bl(static_cast<std::size_t>(numBuses_), Buf(static_cast<std::size_t>(bs)));
        std::vector<Buf> br(static_cast<std::size_t>(numBuses_), Buf(static_cast<std::size_t>(bs)));
        std::vector<std::array<float*, 2>> ptrs(static_cast<std::size_t>(numBuses_));
        std::vector<AudioBusBuffers> buses(static_cast<std::size_t>(numBuses_));
        for (int b = 0; b < numBuses_; ++b)
        {
            auto ub = static_cast<std::size_t>(b);
            ptrs[ub] = {bl[ub].data(), br[ub].data()};
            buses[ub].numChannels = 2;
            buses[ub].channelBuffers32 = (b == nullBus) ? nullptr : ptrs[ub].data();
            buses[ub].silenceFlags = 0;
        }
        long pos = 0;
        while (pos < total)
        {
            const int n = static_cast<int>(std::min<long>(bs, total - pos));
            if (preBlock) preBlock(pos);
            Krate::Test::EventList el;
            for (const auto& e : evs)
            {
                if (e.pos >= pos && e.pos < pos + n)
                {
                    const auto off = static_cast<int32>(e.pos - pos);
                    if (e.vel > 0.0f) el.addNoteOn(static_cast<int16>(e.pitch), e.vel, off);
                    else el.addNoteOff(static_cast<int16>(e.pitch), off);
                }
            }
            Krate::Test::ParameterChanges pc;
            for (const auto& c : pcs)
                if (c.pos >= pos && c.pos < pos + n) pc.addChange(c.id, c.v);
            for (int b = 0; b < numBuses_; ++b)
            {
                auto ub = static_cast<std::size_t>(b);
                std::fill(bl[ub].begin(), bl[ub].end(), 0.0f);
                std::fill(br[ub].begin(), br[ub].end(), 0.0f);
                buses[ub].silenceFlags = 0;
            }
            ProcessData d{};
            d.processMode = kRealtime;
            d.symbolicSampleSize = kSample32;
            d.numSamples = n;
            d.numOutputs = numBuses_;
            d.outputs = buses.data();
            d.numInputs = 0;
            d.inputs = nullptr;
            d.inputEvents = &el;
            d.inputParameterChanges = pc.getParameterCount() > 0 ? &pc : nullptr;
            p->process(d);
            for (int b = 0; b < numBuses_; ++b)
            {
                auto ub = static_cast<std::size_t>(b);
                if (b != nullBus)
                {
                    std::copy(bl[ub].begin(), bl[ub].begin() + n, r.L[ub].begin() + pos);
                    std::copy(br[ub].begin(), br[ub].begin() + n, r.R[ub].begin() + pos);
                }
                r.busFlags[ub].push_back(buses[ub].silenceFlags);
            }
            r.mainFlags.push_back(buses[0].silenceFlags);
            r.activeVoices.push_back(p->voicePoolForTest().getActiveVoiceCount());
            pos += n;
        }
        return r;
    }

    std::unique_ptr<Membrum::Processor> p;

private:
    double sr_;
    int maxBlock_;
    int numBuses_;
};

// ---------------------------------------------------------------- state helpers

{
    Steinberg::MemoryStream s;
    p.getState(&s);
    return std::vector<char>(s.getData(), s.getData() + s.getSize());

{
    Steinberg::MemoryStream s(const_cast<char*>(b.data()), static_cast<TSize>(b.size()));
    return p.setState(&s);
}

bool decodeKit(const std::vector<char>& b, Membrum::State::KitSnapshot& k)
{
    Steinberg::MemoryStream s(const_cast<char*>(b.data()), static_cast<TSize>(b.size()));
    return Membrum::State::readKitBlob(&s, k) == kResultOk;
}

std::vector<char> readAll(IBStream* s)
{
    std::vector<char> out;
    s->seek(0, IBStream::kIBSeekSet, nullptr);
    std::array<char, 4096> chunk{};
    int32 got = 0;
    while (s->read(chunk.data(), static_cast<int32>(chunk.size()), &got) == kResultOk && got > 0)
        out.insert(out.end(), chunk.data(), chunk.data() + got);
    return out;
}

tresult sendBytesMessage(Membrum::Processor& p, const char* id, const std::vector<char>& b, int pad = -1)
{
    auto msg = owned(new HostMessage());
    msg->setMessageID(id);
    auto* a = msg->getAttributes();
    a->setBinary("bytes", b.data(), static_cast<uint32>(b.size()));
    if (pad >= 0) a->setInt("pad", pad);
    return p.notify(msg);
}

// Field-level diff between two kits; returns map name -> number of pads differing.
std::map<std::string, int> diffKits(const Membrum::State::KitSnapshot& a, const Membrum::State::KitSnapshot& b)
{
    std::map<std::string, int> d;
    if (a.maxPolyphony != b.maxPolyphony) d["maxPolyphony"]++;
    if (a.voiceStealingPolicy != b.voiceStealingPolicy) d["stealing"]++;
    if (a.globalCoupling != b.globalCoupling) d["globalCoupling"]++;
    if (a.snareBuzz != b.snareBuzz) d["snareBuzz"]++;
    if (a.tomResonance != b.tomResonance) d["tomResonance"]++;
    if (a.couplingDelayMs != b.couplingDelayMs) d["couplingDelay"]++;
    if (a.masterGainNorm != b.masterGainNorm) d["masterGain"]++;
    if (a.selectedPadIndex != b.selectedPadIndex) d["selectedPad"]++;
    for (std::size_t p = 0; p < a.pads.size(); ++p)
    {
        const auto& x = a.pads[p];
        const auto& y = b.pads[p];
        if (x.exciterType != y.exciterType) d["exciterType"]++;
        if (x.bodyModel != y.bodyModel) d["bodyModel"]++;
        for (std::size_t s = 0; s < x.sound.size(); ++s)
            if (std::memcmp(&x.sound[s], &y.sound[s], sizeof(double)) != 0) d["slot" + std::to_string(s)]++;
        if (x.chokeGroup != y.chokeGroup) d["choke"]++;
        if (x.outputBus != y.outputBus) d["bus"]++;
        if (x.couplingAmount != y.couplingAmount) d["couplingAmount"]++;
        for (std::size_t m = 0; m < 5; ++m)
            if (x.macros[m] != y.macros[m]) d["macro" + std::to_string(m)]++;
    }
    return d;
}

std::string diffStr(const std::map<std::string, int>& d)
{
    if (d.empty()) return "none";
    std::string s;
    for (const auto& [k, v] : d)
        s += k + "x" + std::to_string(v) + " ";
    return s;
}

std::vector<std::uint8_t> readFileBytes(const std::filesystem::path& p)
{
    std::ifstream in(p, std::ios::binary);
    if (!in) return {};
    in.seekg(0, std::ios::end);
    const auto size = static_cast<std::streamsize>(in.tellg());
    in.seekg(0, std::ios::beg);
    std::vector<std::uint8_t> buf(static_cast<std::size_t>(size));
    if (size > 0) in.read(reinterpret_cast<char*>(buf.data()), size);
    return buf;
}

struct FactoryKit
{
    std::string name;
    std::vector<char> component;
};

std::vector<FactoryKit> factoryKits()
{
    std::vector<FactoryKit> kits;
    const std::filesystem::path root =
        std::filesystem::path(MEMBRUM_RESOURCES_DIR) / "presets" / "Kit Presets";
    if (!std::filesystem::exists(root)) return kits;
    std::vector<std::filesystem::path> files;
    for (const auto& e : std::filesystem::recursive_directory_iterator(root))
        if (e.is_regular_file() && e.path().extension() == ".vstpreset") files.push_back(e.path());
    std::sort(files.begin(), files.end());
    for (const auto& f : files)
    {
        auto bytes = readFileBytes(f);
        auto ps = owned(new MemoryStream(bytes.data(), static_cast<TSize>(bytes.size())));
        PresetFile pf(ps);
        if (!pf.readChunkList()) continue;
        const auto* ce = pf.getEntry(kComponentState);
        if (!ce) continue;
        std::vector<char> comp(static_cast<std::size_t>(ce->size));
        ps->seek(ce->offset, IBStream::kIBSeekSet, nullptr);
        int32 got = 0;
        ps->read(comp.data(), static_cast<int32>(comp.size()), &got);
        kits.push_back({f.parent_path().filename().string() + "/" + f.stem().string(), comp});
    }
    return kits;
}

    std::vector<std::pair<ParamID, double>> edits;

        edits.emplace_back(id, v);
        return kResultOk;
    }
    tresult PLUGIN_API endEdit(ParamID) override { ++ends; return kResultOk; }
    tresult PLUGIN_API restartComponent(int32) override { return kResultOk; }
    tresult PLUGIN_API queryInterface(const TUID, void** obj) override
    {
        *obj = nullptr;
        return kNoInterface;
    }
    uint32 PLUGIN_API addRef() override { return 1; }

};

ParamID padId(int pad, int off) { return static_cast<ParamID>(Membrum::padParamId(pad, off)); }

{
    std::vector<Pc> v;
    v.reserve(e.size());
    for (const auto& [id, val] : e) v.push_back({pos, id, val});
    return v;
}

// 1. Exciter x body matrix
// ==============================================================================
static void runMatrix(int exFrom, int exTo)
{
    const double sr = 48000.0;
    for (int e = exFrom; e < exTo; ++e)
    {
        for (int b = 0; b < 6; ++b)
        {
            PadConfig c;
            c.exciterType = static_cast<ExciterType>(e);
            c.bodyModel = static_cast<BodyModelType>(b);
            if (c.exciterType == ExciterType::Friction) c.frictionPressure = 0.5f;
            const Buf hi = renderVoice(c, 1.0f, sr, static_cast<long>(sr * 10.0));
            const Buf lo = renderVoice(c, 0.25f, sr, static_cast<long>(sr * 2.0));
            const auto sh = stats(hi);
            const auto sl = stats(lo);
            const double d60 = decay60(hi, sr);
            const double cen = centroid(hi, sr, 0, 8192);
            const double velDb = toDb(sh.peak) - toDb(sl.peak);
            const std::string nm = std::string("matrix:exciter=") + exName(e) + ":body=" + bodyName(b);
            const std::string setup = "DrumVoice 48k PadConfig defaults, vel 1.0 vs 0.25";
            const bool sustain = c.exciterType == ExciterType::Friction || c.exciterType == ExciterType::Feedback;
            probe(nm, setup, "peak dBFS / rms dBFS / dc", fmt(toDb(sh.peak)) + " / " + fmt(toDb(sh.rms)) + " / " + fmt(sh.dc, 6),
                  "peak <= 0 dBFS, |dc| < 1e-3", (sh.peak > 1.0 || std::fabs(sh.dc) > 1e-3) ? "suspicious" : "ok");
            probe(nm, setup, "nonFinite", std::to_string(sh.nonFinite + sl.nonFinite), "0", (sh.nonFinite + sl.nonFinite) ? "bad" : "ok");
            probe(nm, setup, "decay -60 dB", decayStr(d60, 10.0), sustain ? "sustain expected (Friction/Feedback)" : "< 10 s",
                  (d60 < 0.0 && !sustain) ? "suspicious" : "ok");
            probe(nm, setup, "centroid first 8192 samples Hz", fmt(cen, 0), "info", "ok");
            probe(nm, setup, "peak(vel1.0) - peak(vel0.25) dB", fmt(velDb), ">= 3 dB (else flattened dynamics)",
                  velDb < 3.0 ? "suspicious" : "ok");
            CHECK(fin(sh.peak));
        }
    }
}

// 2. Mode accuracy
// ==============================================================================
TEST_CASE("audit probe 02 mode accuracy", "[.][audit-probe]")
{
    const double sr = 48000.0;
    const BodyModelType bodies[] = {BodyModelType::Membrane, BodyModelType::Plate, BodyModelType::Shell, BodyModelType::Bell};
    const float sizes[] = {0.0f, 0.5f, 1.0f};
    for (auto body : bodies)
    {
        const float* ratios = nullptr;
        int nr = 0;
        switch (body)
        {
        case BodyModelType::Membrane: ratios = Membrum::kMembraneRatios.data(); nr = static_cast<int>(Membrum::kMembraneRatios.size()); break;
        case BodyModelType::Plate: ratios = Membrum::Bodies::kPlateRatios; nr = Membrum::Bodies::kPlateMaxModeCount; break;
        case BodyModelType::Shell: ratios = Membrum::Bodies::kShellRatios; nr = Membrum::Bodies::kShellModeCount; break;
        default: ratios = Membrum::Bodies::kBellRatios; nr = Membrum::Bodies::kBellModeCount; break;
        }
        for (float sz : sizes)
        {
            PadConfig c = detPad(body, ExciterType::Impulse, sz);
            c.airLoading = 0.0f;
            c.modeScatter = 0.0f;
            c.decay = 0.7f;
            c.strikePosition = 0.3f;
            std::array<float, 8> bankF{};
            int nModes = 0;
            const Buf x = renderVoice(c, 1.0f, sr, static_cast<long>(sr * 1.5), 256, -1,
                                      [&](Membrum::DrumVoice& v, long pos) {
                                          if (pos == 256)
                                          {
                                              auto& bank = v.getBodyBankForTest().getSharedBank();
                                              nModes = bank.getNumModes();
                                              for (int k = 0; k < 8; ++k) bankF[static_cast<std::size_t>(k)] = bank.getModeFrequency(k);
                                          }
                                      });
            const double f0 = static_cast<double>(Membrum::Bodies::naturalFundamentalHz(body, sz));
            const std::size_t start = static_cast<std::size_t>(sr * 0.005);
            const std::size_t n = std::min<std::size_t>(x.size() - start, static_cast<std::size_t>(sr * 0.6));
            const auto w = hann(n);
            double maxErrTable = 0.0;
            double maxErrBank = 0.0;
            double strongest = 0.0;
            std::array<double, 8> meas{};
            std::array<double, 8> mags{};
            const int nk = std::min(8, nr);
            for (int k = 0; k < nk; ++k)
            {
                const double target = bankF[static_cast<std::size_t>(k)] > 0.0f ? bankF[static_cast<std::size_t>(k)] : f0 * ratios[k];
                if (target >= sr * 0.5) { meas[static_cast<std::size_t>(k)] = 0.0; continue; }
                double mg = 0.0;
                meas[static_cast<std::size_t>(k)] = refinePeak(x.data() + start, w, sr, target, 40.0, &mg);
                mags[static_cast<std::size_t>(k)] = mg;
                strongest = std::max(strongest, mg);
            }
            std::string detail;
            for (int k = 0; k < nk; ++k)
            {
                const auto uk = static_cast<std::size_t>(k);
                const double table = f0 * ratios[k];
                const bool present = mags[uk] > strongest * 0.003;  // > -50 dB rel
                const double eT = cents(meas[uk], table);
                const double eB = cents(meas[uk], bankF[uk]);
                if (present && meas[uk] > 0.0)
                {
                    maxErrTable = std::max(maxErrTable, std::fabs(eT));
                    maxErrBank = std::max(maxErrBank, std::fabs(eB));
                }
                detail += "k" + std::to_string(k) + ":tab=" + fmt(table, 1) + ",bank=" + fmt(bankF[uk], 1) + ",meas=" +
                          (present ? fmt(meas[uk], 1) : std::string("absent")) + ",dT=" + fmt(eT, 1) + "c ";
            }
            const std::string nm = std::string("modes:body=") + bodyName(static_cast<int>(body)) + ":size=" + fmt(sz, 1);
            const std::string setup = "DrumVoice 48k Impulse, noise/click 0, airLoading 0, scatter 0, decay 0.7";
            probe(nm, setup, "partials (table/bank/measured Hz, cents vs table)", detail, "info", "ok");
            probe(nm, setup, "max |cents| measured vs table (first 8)", fmt(maxErrTable, 1), "< 5 cents",
                  maxErrTable > 5.0 ? "suspicious" : "ok");
            probe(nm, setup, "max |cents| measured vs installed bank freq", fmt(maxErrBank, 1), "< 5 cents",
                  maxErrBank > 5.0 ? "suspicious" : "ok");
            const double fund = meas[0];
            probe(nm, setup, "mode0 measured Hz vs naturalFundamentalHz", fmt(fund, 2) + " vs " + fmt(f0, 2) + " (" + fmt(cents(fund, f0), 1) + " c)",
                  "< 5 cents; plugin_ids.h documents 500 Hz..50 Hz (Membrane)", std::fabs(cents(fund, f0)) > 5.0 ? "suspicious" : "ok");
            probe(nm, setup, "installed mode count", std::to_string(nModes), "info", "ok");
            CHECK(fin(maxErrTable));
        }
    }
    // String body
    for (float sz : sizes)
    {
        PadConfig c = detPad(BodyModelType::String, ExciterType::Impulse, sz);
        c.decay = 0.7f;
        const Buf x = renderVoice(c, 1.0f, sr, static_cast<long>(sr * 1.5));
        const double f0 = static_cast<double>(Membrum::Bodies::naturalFundamentalHz(BodyModelType::String, sz));
        const std::size_t start = static_cast<std::size_t>(sr * 0.02);
        const auto w = hann(static_cast<std::size_t>(sr * 0.6));
        double mg = 0.0;
        const double meas = refinePeak(x.data() + start, w, sr, f0, 100.0, &mg);
        const double dom = dominantPeak(x, sr, start, static_cast<std::size_t>(sr * 0.6), 20.0, 5000.0);
        probe("modes:body=String:size=" + fmt(sz, 1), "DrumVoice 48k Impulse, noise/click 0, decay 0.7",
              "fundamental Hz near f0 / dominant Hz / f0", fmt(meas, 2) + " / " + fmt(dom, 2) + " / " + fmt(f0, 2) + " (" + fmt(cents(meas, f0), 1) + " c)",
              "< 5 cents", std::fabs(cents(meas, f0)) > 5.0 ? "suspicious" : "ok");
        CHECK(fin(meas));
    }
}

// 3. Sample-rate consistency + Nyquist at 44.1 k size 0
// ==============================================================================
TEST_CASE("audit probe 03a sample-rate consistency", "[.][audit-probe]")
{
    const double rates[] = {44100.0, 48000.0, 96000.0, 192000.0};
    const int pads[] = {36, 38, 49};
    const char* names[] = {"kick36", "snare38", "crash49"};
    for (int pi = 0; pi < 3; ++pi)
    {
        std::array<double, 4> f{}, dec{}, rmsDb{}, cen{};
        int nonFin = 0;
        for (int ri = 0; ri < 4; ++ri)
        {
            const double sr = rates[ri];
            Host h(sr, 512);
            const long total = static_cast<long>(sr * 4.0);
            auto r = h.render(total, 512, {{0, pads[pi], 1.0f}});
            const Buf& x = r.L[0];
            const auto s = stats(x);
            nonFin += s.nonFinite;
            f[static_cast<std::size_t>(ri)] = dominantPeak(x, sr, static_cast<std::size_t>(sr * 0.08), static_cast<std::size_t>(sr * 0.5), 20.0, 4000.0);
            dec[static_cast<std::size_t>(ri)] = decay60(x, sr);
            rmsDb[static_cast<std::size_t>(ri)] = toDb(stats(x, 0, static_cast<std::size_t>(sr * 1.0)).rms);
            // centroid over the first 40 ms (same duration at every rate)
            cen[static_cast<std::size_t>(ri)] = centroid(x, sr, 0, static_cast<std::size_t>(sr * 0.04));
        }
        double maxC = 0.0, maxD = 0.0, maxR = 0.0, maxCen = 0.0;
        for (int ri = 0; ri < 4; ++ri)
        {
            const auto u = static_cast<std::size_t>(ri);
            maxC = std::max(maxC, std::fabs(cents(f[u], f[1])));
            if (dec[u] > 0 && dec[1] > 0) maxD = std::max(maxD, std::fabs(dec[u] / dec[1] - 1.0) * 100.0);
            maxR = std::max(maxR, std::fabs(rmsDb[u] - rmsDb[1]));
            if (cen[1] > 0) maxCen = std::max(maxCen, std::fabs(cen[u] / cen[1] - 1.0) * 100.0);
        }
        std::string det;
        for (int ri = 0; ri < 4; ++ri)
        {
            const auto u = static_cast<std::size_t>(ri);
            det += fmt(rates[ri] / 1000.0, 1) + "k:f=" + fmt(f[u], 2) + ",dec=" + decayStr(dec[u], 4.0) + ",rms1s=" + fmt(rmsDb[u]) + ",cen=" + fmt(cen[u], 0) + " ";
        }
        const std::string nm = std::string("srate:") + names[pi];
        const std::string setup = "Processor default kit, vel 1.0, bs 512, 44.1/48/96/192k";
        probe(nm, setup, "per-rate values", det, "info", "ok");
        probe(nm, setup, "max fundamental deviation vs 48k (cents)", fmt(maxC, 1), "<= 5 cents", maxC > 5 ? "bad" : "ok");
        probe(nm, setup, "max decay deviation vs 48k (%)", fmt(maxD, 1), "<= 10 %", maxD > 10 ? "bad" : "ok");
        probe(nm, setup, "max RMS(1 s) deviation vs 48k (dB)", fmt(maxR), "<= 1 dB", maxR > 1 ? "bad" : "ok");
        probe(nm, setup, "max centroid(40 ms) deviation vs 48k (%)", fmt(maxCen, 1), "<= 10 % (info: band-limit differs)", maxCen > 10 ? "suspicious" : "ok");
        probe(nm, setup, "nonFinite", std::to_string(nonFin), "0", nonFin ? "bad" : "ok");
        CHECK(fin(maxC));
    }
}

TEST_CASE("audit probe 04b block-size invariance", "[.][audit-probe]")
{
    const double sr = 48000.0;
    const int sizes[] = {32, 64, 512, 2048, 16384};
    const std::vector<Ev> evs = {{0, 36, 1.0f}, {3001, 38, 0.8f}, {7003, 41, 0.9f}, {12345, 43, 0.7f},
                                 {20000, 36, 0.6f}, {26011, 38, 1.0f}, {33333, 41, 0.5f}, {40000, 43, 0.9f}};
    auto configure = [](Host& h) {
        h.p->voicePoolForTest().padConfigMut(0) = detPad(BodyModelType::Membrane, ExciterType::Impulse, 0.7f);
        h.p->voicePoolForTest().padConfigMut(2) = detPad(BodyModelType::Plate, ExciterType::Mallet, 0.4f);
        h.p->voicePoolForTest().padConfigMut(5) = detPad(BodyModelType::Shell, ExciterType::Impulse, 0.5f);
        h.p->voicePoolForTest().padConfigMut(7) = detPad(BodyModelType::Bell, ExciterType::Mallet, 0.3f);
    };
    const long total = 60000;
    std::vector<Buf> full;
    for (int bs : sizes)
    {
        Host h(sr, bs);
        configure(h);
        full.push_back(h.render(total, bs, evs).L[0]);
        // per-hit onset on isolated renders
        std::string on;
        long worst = 0;
        for (const auto& e : evs)
        {
            Host hi(sr, bs);
            configure(hi);
            auto r = hi.render(e.pos + 4096, bs, {e});
            const long first = firstAbove(r.L[0], 1e-9);
            on += std::to_string(first - e.pos) + " ";
            worst = std::max(worst, std::labs(first - e.pos));
        }
        probe("blocksize:bs=" + std::to_string(bs), "Processor 48k, 8 hits / 4 deterministic pads",
              "onset - event position per hit (samples)", on, "0 for every hit", worst != 0 ? "bad" : "ok");
    }
    for (std::size_t i = 1; i < full.size(); ++i)
    {
        const double d = maxAbsDiff(full[0], full[i]);
        probe("blocksize:bs=" + std::to_string(sizes[i]) + "_vs_32", "Processor 48k, same 8-hit stream",
              "max |sample diff| vs bs 32 render (dBFS)", fmt(toDb(d)), "< -90 dBFS (block-size independent)",
              d > 3e-5 ? "bad" : "ok");
        CHECK(fin(d));
    }
}

TEST_CASE("audit probe 05a roll + same-slot steals", "[.][audit-probe]")
{
    const double sr = 48000.0;

    {
        Host h(sr, 256);

        const double firstPk = stats(x, 0, 480).peak;
        const double lastPk = stats(x, 19 * 480, 20 * 480).peak;
        double maxPk = 0.0;
        for (int i = 0; i < 20; ++i) maxPk = std::max(maxPk, stats(x, static_cast<std::size_t>(i * 480), static_cast<std::size_t>((i + 1) * 480)).peak);

              "ratio < 1.5", rollJump > 1.5 * singleJump ? "suspicious" : "ok");
    }
    // Three steals of the same slot inside 5 ms
    {
        Host h(sr, 32);
        h.p->voicePoolForTest().padConfigMut(0) = detPad(BodyModelType::Membrane, ExciterType::Mallet, 0.6f);
        auto r = h.render(static_cast<long>(sr * 0.5), 32, {{0, 36, 1.0f}, {72, 36, 1.0f}, {144, 36, 1.0f}, {216, 36, 1.0f}});
        const auto s = stats(r.L[0]);
        probe("lifecycle:triple-steal-5ms", "Processor 48k bs 32, same pad at 0/1.5/3/4.5 ms",
              "peak dBFS / max jump / nonFinite", fmt(toDb(s.peak)) + " / " + fmt(maxJump(r.L[0]), 4) + " / " + std::to_string(s.nonFinite),
              "finite, peak <= -1 dBTP limiter", s.nonFinite ? "bad" : "ok");

    }
}

    const char* pol[] = {"Oldest", "Quietest", "Priority"};
    for (int poly : polys)
    {
        for (int policy = 0; policy < 3; ++policy)
        {
            Host h(sr, 64);
            for (int p = 0; p < 32; ++p)
                h.p->voicePoolForTest().padConfigMut(p) = detPad(BodyModelType::Membrane, ExciterType::Mallet, 0.3f + 0.02f * static_cast<float>(p));
            std::vector<Pc> pcs = {{0, Membrum::kMaxPolyphonyId, static_cast<double>(poly - 4) / 12.0},
                                   {0, Membrum::kVoiceStealingId, (policy + 0.5) / 3.0}};
            std::vector<Ev> evs;
            const long spacing = 960;  // 20 ms
            for (int i = 0; i < poly + 4; ++i) evs.push_back({64 + i * spacing, 36 + i, 0.9f});
            auto r = h.render(64 + (poly + 6) * spacing, 64, evs, pcs);
            const Buf& x = r.L[0];
            double jNo = 0.0, jSteal = 0.0;
            for (int i = 0; i < poly + 4; ++i)
            {
                const std::size_t a = static_cast<std::size_t>(64 + i * spacing);
                const double j = maxJump(x, a > 64 ? a - 64 : 0, a + 64);
                if (i < poly) jNo = std::max(jNo, j); else jSteal = std::max(jSteal, j);
            }
            const int clicks = clickCount(x, sr);
            const int maxActive = *std::max_element(r.activeVoices.begin(), r.activeVoices.end());
            probe("lifecycle:steal:poly=" + std::to_string(poly) + ":policy=" + pol[policy],
                  "Processor 48k bs 64, poly+4 deterministic pads 20 ms apart",
                  "max jump at steal hits / at non-steal hits / clicks / max active voices",
                  fmt(jSteal, 4) + " / " + fmt(jNo, 4) + " / " + std::to_string(clicks) + " / " + std::to_string(maxActive),
                  "steal jump <= 1.5x non-steal jump; active <= poly",
                  (jSteal > 1.5 * jNo || maxActive > poly) ? "suspicious" : "ok");
            CHECK(fin(jSteal));
        }
    }
}

static void runTails(BodyModelType body)
{
    const double sr = 48000.0;
    for (int e = 0; e < 7; ++e)
    {
        Host h(sr, 512);
        PadConfig c;
        c.bodyModel = body;
        c.exciterType = static_cast<ExciterType>(e);
        if (c.exciterType == ExciterType::Friction) c.frictionPressure = 0.5f;
        h.p->voicePoolForTest().padConfigMut(0) = c;
        auto r = h.render(static_cast<long>(sr * 30.0), 512, {{0, 36, 1.0f}});
        const double t100 = lastAbove(r.L[0], sr, 1e-5);
        const int act = r.activeVoices.back();
        int firstZero = -1;
        for (std::size_t i = 0; i < r.activeVoices.size(); ++i)
            if (r.activeVoices[i] == 0) { firstZero = static_cast<int>(i); break; }
        const bool sustain = c.exciterType == ExciterType::Friction || c.exciterType == ExciterType::Feedback;
        probe(std::string("tail:body=") + bodyName(static_cast<int>(body)) + ":exciter=" + exName(e),
              "Processor 48k bs 512, PadConfig defaults, no note-off, 30 s",
              "time until < -100 dBFS / active voices at end / voice retired at",
              (t100 < 0 ? std::string(">30 s") : fmt(t100, 2) + " s") + " / " + std::to_string(act) + " / " +
                  (firstZero < 0 ? std::string("never") : fmt(firstZero * 512.0 / sr, 2) + " s"),
              sustain ? "sustain is intended for Friction/Feedback" : "retires (< 30 s)",
              (!sustain && (t100 < 0 || act != 0)) ? "suspicious" : "ok");
        CHECK(fin(t100));
    }
}

TEST_CASE("audit probe 06c sustain stoppers + note length", "[.][audit-probe]")
{

    struct Case { const char* name; ExciterType ex; float inject; };
    const Case cases[] = {{"Friction", ExciterType::Friction, 0.0f}, {"Feedback", ExciterType::Feedback, 0.0f},
                          {"modeInject0.5", ExciterType::Impulse, 0.5f}};
    for (const auto& cs : cases)
    {
        auto setup = [&](Host& h) {
            PadConfig c;
            c.exciterType = cs.ex;
            if (cs.ex == ExciterType::Friction) c.frictionPressure = 0.5f;

        };

            return getStateBytes(*tmp.p);
        }();

        const double db = decay60(rb.L[0], sr);

              "decay60 no note-off / note-off at 10 ms / ratio", decayStr(da, 6) + " / " + decayStr(db, 6) + " / " + (da > 0 && db > 0 ? fmt(db / da, 3) : std::string("n/a")),
              "info: a drum hit should barely depend on note length (ratio ~1)",

TEST_CASE("audit probe 07b bus limiter, master gain, pan", "[.][audit-probe]")
{

    }
    // single kick at vel 1.0: limiter involvement

TEST_CASE("audit probe 10c pitch envelope staircase", "[.][audit-probe]")
{
    const double sr = 48000.0;
    const long total = static_cast<long>(sr * 0.12);
    for (int b = 0; b < 6; ++b)
    {
        if (b == static_cast<int>(BodyModelType::String)) continue;
        for (int knee = 0; knee < 2; ++knee)
        {
            auto track = [&](int bs) {
                PadConfig c = detPad(static_cast<BodyModelType>(b));
                c.tsPitchEnvStart = static_cast<float>(hzToPitchNorm(160.0));
                c.tsPitchEnvEnd = static_cast<float>(hzToPitchNorm(50.0));
                c.tsPitchEnvTime = 0.1f;  // 50 ms
                c.tsPitchEnvKneeEnabled = static_cast<float>(knee);
                c.tsPitchEnvMidPitch = static_cast<float>(hzToPitchNorm(120.0));
                std::vector<double> f(static_cast<std::size_t>(total), 0.0);
                float cur = 0.0f;
                // Render while sampling the bank after each block.
                auto v = std::make_unique<Membrum::DrumVoice>();
                v->prepare(sr, 0u);
                Membrum::applyPadConfigToVoice(*v, c);
                v->noteOn(1.0f);
                Buf out(static_cast<std::size_t>(bs));
                for (long pos = 0; pos < total; pos += bs)
                {
                    const int n = static_cast<int>(std::min<long>(bs, total - pos));
                    v->processBlock(out.data(), n);
                    cur = v->getBodyBankForTest().getSharedBank().getModeFrequency(0);
                    for (int i = 0; i < n; ++i) f[static_cast<std::size_t>(pos + i)] = cur;
                }
                return f;
            };
            const auto ref = track(8);
            const double fEnd = ref.back();
            auto at = [&](const std::vector<double>& f, double ms) {
                return f[std::min<std::size_t>(f.size() - 1, static_cast<std::size_t>(sr * ms / 1000.0))];
            };
            for (int bs : {64, 1024})
            {
                const auto f = track(bs);
                double maxErr = 0.0;
                double maxStep = 0.0;
                for (std::size_t i = 1; i < static_cast<std::size_t>(sr * 0.05); ++i)
                {
                    if (f[i] > 0 && ref[i] > 0) maxErr = std::max(maxErr, std::fabs(cents(f[i], ref[i])));
                    if (f[i] > 0 && f[i - 1] > 0) maxStep = std::max(maxStep, std::fabs(cents(f[i], f[i - 1])));
                }
                probe(std::string("pitchenv:body=") + bodyName(b) + ":knee=" + std::to_string(knee) + ":bs=" + std::to_string(bs),
                      "DrumVoice 48k, 160->50 Hz in 50 ms (mid 120 Hz if knee), bank mode0 per block",
                      "mode0 rel end at 0/25/100 ms (target 3.20/-/1.00) / max err vs bs8 (c) / max step (c)",
                      fmt(at(f, 0.5) / fEnd, 3) + "/" + fmt(at(f, 25) / fEnd, 3) + "/" + fmt(at(f, 100) / fEnd, 3) + " / " + fmt(maxErr, 0) + " / " + fmt(maxStep, 0),
                      "staircase error < 50 cents", maxErr > 50 ? "suspicious" : "ok");
            }
            probe(std::string("pitchenv:body=") + bodyName(b) + ":knee=" + std::to_string(knee) + ":ref",
                  "DrumVoice bs 8", "mode0 Hz at 0.5/25/100 ms", fmt(at(ref, 0.5), 1) + " / " + fmt(at(ref, 25), 1) + " / " + fmt(at(ref, 100), 1),
                  "start ~160 x ratio0, end ~50 x ratio0", "ok");
        }
    }
}

static void runFactoryRoundTrip(std::size_t from, std::size_t to)
{
    const double sr = 48000.0;
    const auto kits = factoryKits();
    if (kits.empty()) { WARN("no factory kits"); return; }
    to = std::min(to, kits.size());
    for (std::size_t k = from; k < to; ++k)
    {
        Host h(sr, 256);
        const tresult rc = setStateBytes(*h.p, kits[k].component);
        const auto s1 = getStateBytes(*h.p);
        const Buf r1 = h.render(static_cast<long>(sr * 0.5), 256, {{0, 36, 1.0f}}).L[0];
        setStateBytes(*h.p, s1);
        const auto s2 = getStateBytes(*h.p);
        const Buf r2 = h.render(static_cast<long>(sr * 0.5), 256, {{0, 36, 1.0f}}).L[0];
        Membrum::State::KitSnapshot kc, k1;
        decodeKit(kits[k].component, kc);
        decodeKit(s1, k1);
        Host hb(sr, 256);
        sendBytesMessage(*hb.p, "LoadPresetState", kits[k].component);
        const auto sb = getStateBytes(*hb.p);
        Membrum::State::KitSnapshot kb;
        decodeKit(sb, kb);
        const auto dAB = diffKits(k1, kb);
        probe("state:factory:" + kits[k].name, "Processor 48k: setState(component) -> getState -> setState -> getState",
              "setState rc / s1==s2 / component vs s1 diff / render pad36 max diff",
              std::to_string(rc) + " / " + (s1 == s2 ? "yes" : "no") + " / " + diffStr(diffKits(kc, k1)) + " / " + fmt(maxAbsDiff(r1, r2), 8),
              "identical bytes, render diff 0 (noise layer may be seeded per voice)", (s1 != s2) ? "bad" : "ok");
        probe("state:pathA-vs-pathB:" + kits[k].name, "setState vs LoadPresetState message, fresh processors",
              "decoded getState diff", diffStr(dAB), "only selectedPad (FR-052 keeps it)",
              (dAB.size() > (dAB.count("selectedPad") ? 1u : 0u)) ? "bad" : "ok");
    }
}

TEST_CASE("audit probe 15b pitch envelope staircase, linear curve", "[.][audit-probe]")
{
    const double sr = 48000.0;
    const long total = static_cast<long>(sr * 0.12);
    for (double timeMs : {20.0, 50.0})
    {
        auto track = [&](int bs) {
            PadConfig c = detPad(BodyModelType::Membrane);
            c.airLoading = 0.0f;
            c.tsPitchEnvStart = static_cast<float>(hzToPitchNorm(160.0));
            c.tsPitchEnvEnd = static_cast<float>(hzToPitchNorm(50.0));
            c.tsPitchEnvTime = static_cast<float>(timeMs / 500.0);
            c.tsPitchEnvCurve = 0.5f;  // linear
            std::vector<double> f(static_cast<std::size_t>(total), 0.0);
            auto v = std::make_unique<Membrum::DrumVoice>();
            v->prepare(sr, 0u);
            Membrum::applyPadConfigToVoice(*v, c);
            v->noteOn(1.0f);
            Buf out(static_cast<std::size_t>(bs));
            for (long pos = 0; pos < total; pos += bs)
            {
                const int n = static_cast<int>(std::min<long>(bs, total - pos));
                v->processBlock(out.data(), n);
                const float cur = v->getBodyBankForTest().getSharedBank().getModeFrequency(0);
                for (int i = 0; i < n; ++i) f[static_cast<std::size_t>(pos + i)] = cur;
            }
            return f;
        };
        const auto ref = track(1);
        for (int bs : {32, 64, 256, 1024})
        {
            const auto f = track(bs);
            double maxErr = 0.0;
            double rmsErr = 0.0;
            const std::size_t nS = static_cast<std::size_t>(sr * timeMs / 1000.0);
            for (std::size_t i = 0; i < nS; ++i)
            {
                const double e = cents(f[i], ref[i]);
                maxErr = std::max(maxErr, std::fabs(e));
                rmsErr += e * e;
            }
            rmsErr = std::sqrt(rmsErr / static_cast<double>(nS));
            probe("pitchenv-linear:time=" + fmt(timeMs, 0) + "ms:bs=" + std::to_string(bs),
                  "DrumVoice 48k Membrane, 160->50 Hz linear curve, bank mode0 vs per-sample (bs 1) reference",
                  "max / rms pitch error during the sweep (cents) / ref at 0, T/2, T Hz",
                  fmt(maxErr, 0) + " / " + fmt(rmsErr, 0) + " / " + fmt(ref[0], 1) + ", " + fmt(ref[nS / 2], 1) + ", " + fmt(ref[nS], 1),
                  "< 50 cents (block size must not reshape the sweep)", maxErr > 50 ? "bad" : "ok");
        }
    }
}

        const double dev = (d[1] > 0 && d[3] > 0) ? (d[3] / d[1] - 1.0) * 100.0 : 0.0;
        probe(std::string("srate-voice:body=") + bodyName(b), "DrumVoice Impulse noise/click 0 size 0.6 decay 0.5, block = 256 samples @48k scaled with rate",
              "decay60, dominant Hz per rate / decay deviation 192k vs 48k (%)", det + "/ " + fmt(dev, 1),
              "<= 10 %", std::fabs(dev) > 10.0 ? "bad" : "ok");
    }
    // Processor, kick with tension + pitch env off, block size scaled with rate (same duration per block)
    {
        std::string det;
        for (int ri = 0; ri < 4; ++ri)
        {
            const double sr = rates[ri];
            const int bs = static_cast<int>(512 * sr / 48000.0);
            Host h(sr, bs);
            auto& cfg = h.p->voicePoolForTest().padConfigMut(0);
            cfg.tensionModAmt = 0.0f;
            cfg.noiseLayerMix = 0.0f;
            cfg.clickLayerMix = 0.0f;
            auto r = h.render(static_cast<long>(sr * 4.0), bs, {{0, 36, 1.0f}});
            det += fmt(sr / 1000.0, 1) + "k:" + decayStr(decay60(r.L[0], sr), 4) + "," +
                   fmt(dominantPeak(r.L[0], sr, static_cast<std::size_t>(sr * 0.08), static_cast<std::size_t>(sr * 0.5), 20.0, 4000.0), 2) + "Hz ";
        }
        probe("srate-kick:tension0+noise/click0:bs-scaled", "Processor default kit pad 36, pitch env ON, block = 10.7 ms at every rate",
              "decay60, fundamental per rate", det, "<= 5 cents, <= 10 %", "ok");
    }
}

TEST_CASE("audit probe 15f state precision and malformed values", "[.][audit-probe]")
{
    const double sr = 48000.0;
    const auto kits = factoryKits();
    if (kits.empty()) { WARN("no factory kits"); return; }
    int kitsWithSemanticDiff = 0;
    double worst = 0.0;
    for (const auto& k : kits)
    {
        Host h(sr, 256);
        setStateBytes(*h.p, k.component);
        Membrum::State::KitSnapshot kc, k1;
        decodeKit(k.component, kc);
        decodeKit(getStateBytes(*h.p), k1);
        double w = 0.0;
        for (std::size_t p = 0; p < 32; ++p)
        {
            for (std::size_t s = 0; s < 59; ++s) w = std::max(w, std::fabs(kc.pads[p].sound[s] - k1.pads[p].sound[s]));
            for (std::size_t m = 0; m < 5; ++m) w = std::max(w, std::fabs(kc.pads[p].macros[m] - k1.pads[p].macros[m]));
            w = std::max(w, std::fabs(kc.pads[p].couplingAmount - k1.pads[p].couplingAmount));
        }
        w = std::max(w, std::fabs(kc.globalCoupling - k1.globalCoupling));
        if (w > 1e-6) ++kitsWithSemanticDiff;
        worst = std::max(worst, w);
    }
    probe("state:factory-component-vs-getState-tolerance", "20 factory kits: setState(component) then getState",
          "kits differing by > 1e-6 / worst |diff|", std::to_string(kitsWithSemanticDiff) + " / " + fmt(worst, 9),
          "only float rounding (< 1e-6)", kitsWithSemanticDiff ? "bad" : "ok");
    // Stored values after malformed slots
    const auto good = kits[0].component;
    struct Bad { const char* name; std::size_t slot; std::uint64_t bits; };
    const Bad bads[] = {{"NaN-slot0-material", 0, 0x7FF8000000000000ULL}, {"NaN-slot2-decay", 2, 0x7FF8000000000000ULL},
                        {"Inf-slot1-size", 1, 0x7FF0000000000000ULL}, {"NaN-slot57-noiseGain", 57, 0x7FF8000000000000ULL},
                        {"NaN-slot42-dampB1", 42, 0x7FF8000000000000ULL}, {"NaN-slot34-noiseMix", 34, 0x7FF8000000000000ULL}};
    for (const auto& b : bads)
    {
        auto blob = good;
        std::memcpy(blob.data() + 20 + b.slot * 8, &b.bits, 8);
        Host h(sr, 256);
        const tresult rc = setStateBytes(*h.p, blob);
        Membrum::State::KitSnapshot k;
        decodeKit(getStateBytes(*h.p), k);
        const double stored = k.pads[0].sound[b.slot];
        auto r = h.render(static_cast<long>(sr * 2.0), 256, {{0, 36, 1.0f}});
        const auto s = stats(r.L[0]);
        const bool storedFinite = Krate::DSP::detail::isFinite(stored);
        probe(std::string("state:malformed-stored:") + b.name, "setState(kit with the slot of pad 0 poisoned), getState, render pad 36 2 s",
              "rc / stored value finite / out peak dBFS / nonFinite / decay60",
              std::to_string(rc) + " / " + (storedFinite ? fmt(stored, 4) : std::string("NON-FINITE")) + " / " + fmt(toDb(s.peak)) + " / " +
                  std::to_string(s.nonFinite) + " / " + decayStr(decay60(r.L[0], sr), 2),
              "sanitised on load (finite stored value), finite output", (!storedFinite || s.nonFinite) ? "bad" : "ok");
    }
}
