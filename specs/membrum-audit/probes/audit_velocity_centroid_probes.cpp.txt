struct Rig
{
    std::unique_ptr<Membrum::Processor> proc;
    std::vector<float> buf;
    std::array<float*, static_cast<std::size_t>(kBuses * 2)> chans{};
    std::array<AudioBusBuffers, static_cast<std::size_t>(kBuses)> bus{};
    ProcessData data{};
    Krate::Test::EventList events;
    Krate::Test::ParameterChanges params;
    Rig()
    {
        proc = std::make_unique<Membrum::Processor>();
        proc->initialize(nullptr);
        ProcessSetup s{};
        s.processMode = kRealtime;
        s.symbolicSampleSize = kSample32;
        s.maxSamplesPerBlock = kBS;
        s.sampleRate = kSR;
        proc->setupProcessing(s);
        proc->setActive(true);
        buf.assign(static_cast<std::size_t>(kBuses * 2 * kBS), 0.0f);
        for (int c = 0; c < kBuses * 2; ++c) chans[c] = buf.data() + c * kBS;
        for (int b = 0; b < kBuses; ++b)
        {
            bus[b].numChannels = 2;
            bus[b].channelBuffers32 = &chans[b * 2];
            bus[b].silenceFlags = 0;
        }
        data.processMode = kRealtime;
        data.symbolicSampleSize = kSample32;
        data.numSamples = kBS;
        data.numOutputs = kBuses;
        data.outputs = bus.data();
        data.inputParameterChanges = &params;
        data.inputEvents = &events;
    }
    ~Rig() { proc->setActive(false); proc->terminate(); }
};

std::vector<std::uint8_t> loadComp(const KitRef& k)
{
    const auto path = std::filesystem::path(MEMBRUM_RESOURCES_DIR) / "presets" / "Kit Presets" / k.cat /
                      (std::string(k.name) + ".vstpreset");
    std::ifstream in(path, std::ios::binary);
    std::vector<std::uint8_t> bytes((std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
    if (bytes.empty()) return {};
    auto ps = owned(new MemoryStream(bytes.data(), static_cast<TSize>(bytes.size())));
    ps->seek(0, IBStream::kIBSeekSet, nullptr);
    PresetFile pf(ps);
    if (!pf.readChunkList()) return {};
    const auto* e = pf.getEntry(kComponentState);
    if (!e) return {};
    std::vector<std::uint8_t> comp(static_cast<std::size_t>(e->size));
    ps->seek(e->offset, IBStream::kIBSeekSet, nullptr);
    int32 got = 0;
    ps->read(comp.data(), static_cast<int32>(comp.size()), &got);
    return comp;
}

void loadInto(Rig& rig, const std::vector<std::uint8_t>& comp)
{
    rig.proc->setActive(false);
    MemoryStream ms;
    int32 w = 0;
    ms.write(const_cast<std::uint8_t*>(comp.data()), static_cast<int32>(comp.size()), &w);
    ms.seek(0, IBStream::kIBSeekSet, nullptr);
    rig.proc->setState(&ms);
    rig.proc->setActive(true);
}

std::vector<float> renderMono(Rig& rig, int note, float vel, int n, float* peakOut)
{
    std::vector<float> mono(static_cast<std::size_t>(n), 0.0f);
    float pk = 0.0f;
    for (int b = 0; b < n; b += kBS)
    {
        const int ns = std::min(kBS, n - b);
        rig.events.clear();
        rig.params.clear();
        if (b == 0) rig.events.addNoteOn(static_cast<int16>(note), vel, 0);
        std::fill(rig.buf.begin(), rig.buf.end(), 0.0f);
        rig.data.numSamples = ns;
        rig.proc->process(rig.data);
        for (int s = 0; s < ns; ++s)
        {
            float l = rig.chans[0][s], r = rig.chans[1][s];
            if (!Krate::DSP::detail::isFinite(l)) l = 0.0f;
            if (!Krate::DSP::detail::isFinite(r)) r = 0.0f;
            mono[b + s] = 0.5f * (l + r);
            pk = std::max(pk, std::max(std::abs(l), std::abs(r)));
        }
    }
    if (peakOut) *peakOut = pk;
    return mono;
}

double centroid200(const std::vector<float>& x)
{
    const int fftSize = 4096, hop = 1024;
    const int n = std::min(static_cast<int>(kSR * 0.2), static_cast<int>(x.size()));
    Krate::DSP::FFT fft;
    fft.prepare(fftSize);
    std::vector<float> win(fftSize), frame(fftSize);
    for (int i = 0; i < fftSize; ++i) win[i] = 0.5f * (1.0f - std::cos(2.0f * 3.14159265358979f * i / fftSize));
    std::vector<Krate::DSP::Complex> spec(fftSize / 2 + 1);
    std::vector<double> acc(fftSize / 2 + 1, 0.0);
    for (int start = -fftSize / 2; start < n - fftSize / 2 + hop; start += hop)
    {
        for (int i = 0; i < fftSize; ++i)
        {
            const int idx = start + i;
            frame[i] = ((idx >= 0 && idx < n) ? x[idx] : 0.0f) * win[i];
        }
        fft.forward(frame.data(), spec.data());
        for (std::size_t k = 0; k < spec.size(); ++k) acc[k] += spec[k].magnitude();
    }
    const double binHz = kSR / fftSize;
    double num = 0, den = 0;
    for (std::size_t k = 1; k < acc.size(); ++k) { num += k * binHz * acc[k]; den += acc[k]; }
    return den > 0 ? num / den : 0.0;
}

double rmsDb(const std::vector<float>& x, int n)
{
    n = std::min(n, static_cast<int>(x.size()));
    double ss = 0;
    for (int i = 0; i < n; ++i) ss += double(x[i]) * x[i];
    const double r = std::sqrt(ss / std::max(1, n));
    return r > 1e-9 ? 20 * std::log10(r) : -180.0;
}
double rms500(const std::vector<float>& x) { return rmsDb(x, static_cast<int>(kSR * 0.5)); }
double rms200(const std::vector<float>& x) { return rmsDb(x, static_cast<int>(kSR * 0.2)); }

using Mut = std::function<void(Membrum::PadConfig&)>;

std::vector<float> hit(Rig& rig, const std::vector<std::uint8_t>& comp, int note, float vel, const Mut& m,
                       float* peak)
{
    const int pad = note - 36;
    // warm-up (fills the strike-normalisation cache for this config on slot 0)
    loadInto(rig, comp);

    m(rig.proc->voicePoolForTest().padConfigMut(pad));
    return renderMono(rig, note, vel, static_cast<int>(kSR), peak);
}

    return d;
}

int main()
{
    Rig rig;
    const float vLo = 40.0f / 127.0f, vHi = 1.0f;
    const auto lev = [](Membrum::PadConfig& c) { c.level *= 0.1f; };
    struct Var { const char* name; Mut m; };
    const Mut none = [](Membrum::PadConfig&) {};
    const Mut noNL = [](Membrum::PadConfig& c) { c.noiseLayerMix = 0.0f; };
    const Mut noCL = [](Membrum::PadConfig& c) { c.clickLayerMix = 0.0f; };
    const Mut noNLC = [](Membrum::PadConfig& c) { c.nonlinearCoupling = 0.0f; };
    const Mut noTen = [](Membrum::PadConfig& c) { c.tensionModAmt = 0.0f; };
    const Mut noUN = [](Membrum::PadConfig& c) {
        c.nonlinearCoupling = 0.0f; c.modeInjectAmount = 0.0f; c.tensionModAmt = 0.0f; c.wireCoupling = 0.0f;
        c.secondaryEnabled = 0.0f; };
    const Mut neutralTS = [](Membrum::PadConfig& c) {
        c.tsFilterCutoff = 1.0f; c.tsFilterEnvAmount = 0.5f; c.tsDriveAmount = 0.0f; c.tsFoldAmount = 0.0f;
        c.tsFilterResonance = 0.0f; };
    auto cat = [](std::vector<Mut> ms) { return Mut([ms](Membrum::PadConfig& c) { for (auto& m : ms) m(c); }); };

    const std::vector<Var> vars = {
        {"V0 as-is", none},
        {"V1 level x0.1", lev},
        {"V2 lev+noiseMix0", cat({lev, noNL})},
        {"V3 lev+clickMix0", cat({lev, noCL})},
        {"V4 lev+noise0+click0", cat({lev, noNL, noCL})},
        {"V5 V4+NLC0", cat({lev, noNL, noCL, noNLC})},
        {"V6 V4+UN0(NLC,inject,tension,wire,shell)", cat({lev, noNL, noCL, noUN})},
        {"V7 V6+toneShaperNeutral", cat({lev, noNL, noCL, noUN, neutralTS})},
        {"V8 lev+NLC0 only", cat({lev, noNLC})},
        {"V9 lev+tension0 only", cat({lev, noTen})},
        {"V10 lev+NLC0+tension0", cat({lev, noNLC, noTen})},
    };

    for (const auto& t : kTargets)

int otherPad(const Membrum::PadConfig* cfgs, int pad)
{
    for (int p = 0; p < Membrum::kNumPads; ++p)
        if (p != pad && cfgs[p].enabled >= 0.5f && cfgs[p].exciterType != cfgs[pad].exciterType &&
            cfgs[p].bodyModel != cfgs[pad].bodyModel)
            return p;
    for (int p = 0; p < Membrum::kNumPads; ++p)
        if (p != pad && cfgs[p].enabled >= 0.5f && cfgs[p].exciterType != cfgs[pad].exciterType) return p;
    return pad == 0 ? 1 : 0;
}

// cold hit: load, hit `prevNote`, reload, hit target (slot 0 last played prevNote)
std::vector<float> coldHit(Rig& rig, const std::vector<std::uint8_t>& comp, int prevNote, int note, float vel, float* pk)
{
    loadInto(rig, comp);
    (void)renderMono(rig, prevNote, 1.0f, kBS * 4, nullptr);
    loadInto(rig, comp);
    return renderMono(rig, note, vel, static_cast<int>(kSR), pk);
}
std::vector<float> warmHit(Rig& rig, const std::vector<std::uint8_t>& comp, int note, float vel, float warmVel, float* pk)

    loadInto(rig, comp);
    (void)renderMono(rig, note, warmVel, kBS * 4, nullptr);
    loadInto(rig, comp);
    return renderMono(rig, note, vel, static_cast<int>(kSR), pk);
}

void bandReport(const char* label, const std::vector<double>& x)
{
    // DFT energy in bands (zero padded to 8192)
    const int N = 8192;
    std::vector<float> buf(N, 0.0f);
    for (int i = 0; i < std::min<int>(N, static_cast<int>(x.size())); ++i) buf[i] = static_cast<float>(x[i]);
    Krate::DSP::FFT fft;
    fft.prepare(N);
    std::vector<Krate::DSP::Complex> spec(N / 2 + 1);
    fft.forward(buf.data(), spec.data());
    const double edges[] = {50, 150, 400, 1000, 2500, 6000, 24000};
    double e[6] = {};
    double num = 0, den = 0, numB = 0, denB = 0;
    for (int k = 1; k <= N / 2; ++k)
    {
        const double f = k * kSR / N;
        const double m = spec[k].magnitude();
        for (int b = 0; b < 6; ++b)
            if (f >= edges[b] && f < edges[b + 1]) e[b] += m * m;
        num += f * m; den += m;
        if (f < 2500) { numB += f * m; denB += m; }
    }
    std::printf("    %-34s centroid full %7.1f Hz, <2.5k %6.1f Hz | band dB 50-150 %6.1f 150-400 %6.1f 400-1k %6.1f "
                "1k-2.5k %6.1f 2.5k-6k %6.1f 6k+ %6.1f\n",
                label, den > 0 ? num / den : 0.0, denB > 0 ? numB / denB : 0.0,
                10 * std::log10(e[0] + 1e-30), 10 * std::log10(e[1] + 1e-30), 10 * std::log10(e[2] + 1e-30),
                10 * std::log10(e[3] + 1e-30), 10 * std::log10(e[4] + 1e-30), 10 * std::log10(e[5] + 1e-30));
}

void exciterAnalysis(Membrum::ExciterType type, float sizeHint)
{
    const int n = static_cast<int>(kSR * 0.03);
    for (float v : {40.0f / 127.0f, 1.0f})
    {
        constexpr int kSeeds = 64;
        std::vector<std::vector<double>> runs(kSeeds, std::vector<double>(n, 0.0));
        for (int s = 0; s < kSeeds; ++s)
        {
            Membrum::ExciterBank bank;
            bank.prepare(kSR, static_cast<std::uint32_t>(s));
            bank.setExciterType(type);
            bank.setMalletBodySizeHint(sizeHint);
            bank.trigger(v);
            for (int i = 0; i < n; ++i) runs[s][i] = bank.process(0.0f);
        }
        std::vector<double> mean(n, 0.0);
        for (int i = 0; i < n; ++i) { for (int s = 0; s < kSeeds; ++s) mean[i] += runs[s][i]; mean[i] /= kSeeds; }
        // noise: average band energy of residuals
        std::printf("  %s v=%.3f\n", type == Membrum::ExciterType::Impulse ? "Impulse" : "Mallet", v);
        bandReport("one seed (output)", runs[0]);
        bandReport("pulse (mean of 64 seeds)", mean);
        std::vector<double> res(n);
        for (int i = 0; i < n; ++i) res[i] = runs[0][i] - mean[i];
        bandReport("noise (seed0 - mean)", res);
    }
}
} // namespace

                    var.name, centroid200(first), 20 * std::log10(p1 + 1e-12f), centroid200(second), 20 * std::log10(p2 + 1e-12f),
                    centroid200(third), 20 * std::log10(p3 + 1e-12f));
    }
    return 0;
}

// ======================= FILE: audit_velocity_centroid_probe5.cpp =======================
// audit_velocity_centroid_probe5.cpp -- isolate the ToneShaper state difference behind the
// Drone and Sustain 36 first-hit brightness: ToneShaper alone, white-noise input, Drone 36
// settings (LP, cutoff norm 0.45, res 0.35, env amount 0.62 -> +0.24, drive 0.26, A/D/S/R
// norm 0.2/0.4/0.55/0.65), 200 ms per note.
#include "common.inc"
#include "dsp/tone_shaper.h"
#include <krate/dsp/core/xorshift32.h>

namespace {
struct TSCfg { float type, cut, res, envAmt, a, d, s, r, drive; };

void applyTS(Membrum::ToneShaper& ts, const TSCfg& c)
{
    ts.setFilterType(static_cast<Membrum::ToneShaperFilterType>(std::clamp(static_cast<int>(c.type * 3.0f), 0, 2)));
    ts.setFilterCutoff(20.0f * std::pow(1000.0f, c.cut));
    ts.setFilterResonance(c.res);
    ts.setFilterEnvAmount(c.envAmt * 2.0f - 1.0f);
    ts.setFilterEnvAttackMs(c.a * c.a * c.a * 500.0f);
    ts.setFilterEnvDecayMs(c.d * c.d * c.d * 2000.0f);
    ts.setFilterEnvSustain(c.s);
    ts.setFilterEnvReleaseMs(c.r * c.r * c.r * 2000.0f);
    ts.setDriveAmount(c.drive);
    ts.setFoldAmount(0.0f);
}

std::vector<float> note(Membrum::ToneShaper& ts, float vel)
{
    ts.noteOn(vel);
    Krate::DSP::XorShift32 rng;
    rng.seed(1234u);
    std::vector<float> out(static_cast<std::size_t>(kSR * 0.2));
    for (auto& o : out) o = ts.processSample(0.2f * rng.nextFloatSigned());
    return out;
}
} // namespace

int main()
{
    const TSCfg drone{0.0f, 0.45f, 0.35f, 0.62f, 0.2f, 0.4f, 0.55f, 0.65f, 0.26f};
    for (int dv = 0; dv < 2; ++dv)
    {
        TSCfg c = drone;
        if (dv) c.drive = 0.0f;
        Membrum::ToneShaper ts;
        ts.prepare(kSR);
        applyTS(ts, c);
        auto a = note(ts, 0.315f);
        applyTS(ts, c);
        auto b = note(ts, 0.315f);
        ts.prepare(kSR);
        applyTS(ts, c);
        auto d = note(ts, 0.315f);
        applyTS(ts, c);
        auto e = note(ts, 0.315f);
        std::printf("drive %.2f | 1st after prepare-with-defaults %.1f Hz rms %.2f | 2nd (no prepare) %.1f Hz rms %.2f | "
                    "after re-prepare with cached values %.1f Hz rms %.2f | next (no prepare) %.1f Hz rms %.2f\n",
                    c.drive, centroid200(a), rms200(a), centroid200(b), rms200(b), centroid200(d), rms200(d),
                    centroid200(e), rms200(e));
        std::printf("   getFilterCutoff %.1f Hz\n", ts.getFilterCutoff());
    }
    return 0;
}

// ======================= FILE: audit_velocity_centroid_probe6.cpp =======================
// audit_velocity_centroid_probe6.cpp -- does the main-bus coupling return (SympatheticResonance +
// -20 dBFS energy limiter) or the bus limiter move the centroid? Same warm harness, as-is vs
// level x0.1, each with the kit's Global Coupling and with Global Coupling forced to 0.
#include "common.inc"

namespace {
std::vector<float> renderWithParam(Rig& rig, int note, float vel, int n, bool zeroCoupling, float* pk)
{
    std::vector<float> mono(static_cast<std::size_t>(n), 0.0f);
    float p = 0.0f;
    for (int b = 0; b < n; b += kBS)
    {
        rig.events.clear();
        rig.params.clear();
        if (b == 0)
        {
            rig.events.addNoteOn(static_cast<int16>(note), vel, 0);
            if (zeroCoupling) rig.params.addChange(Membrum::kGlobalCouplingId, 0.0);
        }
        const int ns = std::min(kBS, n - b);
        std::fill(rig.buf.begin(), rig.buf.end(), 0.0f);
        rig.data.numSamples = ns;
        rig.proc->process(rig.data);
        for (int s = 0; s < ns; ++s)
        {
            const float l = rig.chans[0][s], r = rig.chans[1][s];
            mono[b + s] = 0.5f * (l + r);
            p = std::max(p, std::max(std::abs(l), std::abs(r)));
        }
    }
    if (pk) *pk = p;
    return mono;
}
std::vector<float> hitC(Rig& rig, const std::vector<std::uint8_t>& comp, int note, float vel, float levelScale, bool zc, float* pk)
{
    for (int pass = 0; pass < 2; ++pass)
    {
        loadInto(rig, comp);
        rig.proc->voicePoolForTest().padConfigMut(note - 36).level *= levelScale;
        auto r = renderWithParam(rig, note, vel, pass ? static_cast<int>(kSR) : kBS * 4, zc, pk);
        if (pass) return r;
    }
    return {};
}
} // namespace

int main()
{
    std::setvbuf(stdout, nullptr, _IONBF, 0);
    Rig rig;
    for (const auto& t : kTargets)
    {
        if (std::string(t.name) == "Latin Perc") continue;
        const auto comp = loadComp(t);
        Membrum::State::KitSnapshot snap;
        {
            MemoryStream ms;
            int32 w = 0;
            ms.write(const_cast<std::uint8_t*>(comp.data()), static_cast<int32>(comp.size()), &w);
            ms.seek(0, IBStream::kIBSeekSet, nullptr);
            Membrum::State::readKitBlob(&ms, snap);
        }
        loadInto(rig, comp);
        const float amt = rig.proc->voicePoolForTest().padConfig(t.note - 36).couplingAmount;
        std::printf("%s:%d globalCoupling %.3f padCouplingAmount %.3f masterNorm %.3f\n", t.name, t.note, snap.globalCoupling, amt,
                    snap.masterGainNorm);
        for (float ls : {1.0f, 0.1f})
            for (bool zc : {false, true})
            {
                float pl = 0, ph = 0;
                auto lo = hitC(rig, comp, t.note, 40.0f / 127.0f, ls, zc, &pl);
                auto hi = hitC(rig, comp, t.note, 1.0f, ls, zc, &ph);
                std::printf("  level x%.1f coupling %s | cLo %7.1f cHi %7.1f ratio %.3f | peak lo %7.2f hi %7.2f dBFS\n", ls,
                            zc ? "0   " : "kit ", centroid200(lo), centroid200(hi), centroid200(hi) / centroid200(lo),
                            20 * std::log10(pl + 1e-12f), 20 * std::log10(ph + 1e-12f));
            }
        std::fflush(stdout);
    }
    return 0;
}
