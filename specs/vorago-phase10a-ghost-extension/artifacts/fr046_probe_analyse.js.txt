function parse(file) {
  const out = {};
  for (const line of fs.readFileSync(file, 'utf8').split(/\r?\n/)) {
    const m = line.match(/^([AB])\.(\w+) (\S+)$/);
    if (!m) continue;
    const [, fx, key, val] = m;
    out[fx] = out[fx] || { cp: [] };
    if (key.startsWith('cp') && /^cp\d\d$/.test(key)) out[fx].cp[parseInt(key.slice(2), 10)] = parseFloat(val);
    else out[fx][key] = parseFloat(val);
  }
  return out;
}

const legs = process.argv.slice(2);
const acc = {};
for (const leg of legs) {
  const got = parse(leg);
  for (const fx of ['A', 'B']) {
    const r = REF[fx], g = got[fx];
    acc[fx] = acc[fx] || { rms: 0, peak: 0, meanAbs: 0, totalVariation: 0, cpAbs: 0, cpIdx: -1, worstLeg: {} };
    for (const k of ['rms', 'peak', 'meanAbs', 'totalVariation']) {
      const rel = Math.abs(g[k] - r[k]) / Math.max(Math.abs(r[k]), 1e-12);
      if (rel > acc[fx][k]) { acc[fx][k] = rel; acc[fx].worstLeg[k] = leg; }
    }
    for (let i = 0; i < 32; i++) {
      // float32 round-trip of the reference literal, as the C++ comparison does
      const refF = Math.fround(r.cp[i]);
      const e = Math.abs(Math.fround(g.cp[i]) - refF);
      if (e > acc[fx].cpAbs) { acc[fx].cpAbs = e; acc[fx].cpIdx = i; acc[fx].worstLeg.cp = leg; }
    }
    console.log(`${leg}  ${fx}  rms=${(Math.abs(g.rms - r.rms) / Math.abs(r.rms)).toExponential(4)}` +
      ` peak=${(Math.abs(g.peak - r.peak) / Math.abs(r.peak)).toExponential(4)}` +
      ` meanAbs=${(Math.abs(g.meanAbs - r.meanAbs) / Math.abs(r.meanAbs)).toExponential(4)}` +
      ` totVar=${(Math.abs(g.totalVariation - r.totalVariation) / Math.abs(r.totalVariation)).toExponential(4)}`);
  }
}
console.log('\n=== WORST OVER ALL LEGS ===');
for (const fx of ['A', 'B']) {
  const a = acc[fx];
  console.log(`Fixture ${fx}:`);
  for (const k of ['rms', 'peak', 'meanAbs', 'totalVariation'])
    console.log(`  ${k.padEnd(15)} ${a[k].toExponential(4)}   (${a.worstLeg[k]})`);
  console.log(`  worst checkpoint ${a.cpAbs.toExponential(4)} at cp${a.cpIdx}  (${a.worstLeg.cp})`);
  console.log(`  => worst METRIC relative ${Math.max(a.rms, a.peak, a.meanAbs, a.totalVariation).toExponential(4)}`);
}
